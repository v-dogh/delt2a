159 void _render_border(PixelBuffer::View buffer) const
160 {
161 auto& p = static_cast<const Base&>(*this);
162 const BoundingBox bbox{std::min(p.box().width, buffer.width()), std::min(p.box().height, buffer.height())};
163 if (bbox.empty()) return;
164
165 const bool is_border = p.container_options & Base::ContainerOptions::EnableBorder;
166 const bool is_topfix = p.container_options & Base::ContainerOptions::TopFix;
167 const bool is_corners = p.container_options & Base::ContainerOptions::OverrideCorners;
168
169 if (is_border)
170 {
171 const auto& bcv = p.border_vertical_color;
172 const auto& bch = p.border_horizontal_color;
173 const auto bw = std::max(0, p.resolve_units(p.border_width));
174
175 const bool et = !(p.container_options & Base::ContainerOptions::DisableBorderTop);
176 const bool eb =
177 !(p.container_options & Base::ContainerOptions::DisableBorderBottom);
178 const bool el = !(p.container_options & Base::ContainerOptions::DisableBorderLeft);
179 const bool er = !(p.container_options & Base::ContainerOptions::DisableBorderRight);
180
181 // Top
182 if (et)
183 for (int i = 0; i < std::min(bw, bbox.height); i++)
184 for (std::size_t x = 0; x < bbox.width; x++)
185 {
186 buffer.at(x, i).blend(bch);
187 }
188
189 // Bottom
190 if (eb)
191 for (int i = 0; i < std::min(bw, bbox.height); i++)
192 for (std::size_t x = 0; x < bbox.width; x++)
193 {
194 buffer.at(x, bbox.height - 1 - i).blend(bch);
195 }
196

198 if (el)
199 for (int i = 0; i < std::min(bw, bbox.width); i++)
200 for (std::size_t y = is_topfix; y < bbox.height; y++)
201 {

203 }
204

208 for (std::size_t y = is_topfix; y < bbox.height; y++)
