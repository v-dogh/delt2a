9 template<std::size_t First, std::size_t Len, std::size_t Count>
10 void flip(std::bitset<Count>& bs)
11 {
12 if constexpr (First < Count && Len != 0)
13 {
14 constexpr std::size_t clamped = Len > Count - First ? Count - First : Len;
15 static const auto mask = []()
16 {
17 std::bitset<Count> m;
18 for (std::size_t i = 0; i < clamped; i++)
19 m.set(First + i);
20 return m;
21 }();
22
23 bs ^= mask;
24 }
25 }

27 template<SystemScreen::Event Ev, typename... Argv> void SystemScreen::_signal(Argv&&... args)
28 {
29 auto ctx = context();
30 if constexpr (Ev == Event::KeyInput)
31 {
32 ctx->signal<Ev>(ref(*input()));
33 }
34 else if constexpr (Ev == Event::KeySequenceInput)
35 {
36 auto view = in::internal::InputFrameView::from(input());

39 const auto sequence = input()->sequence();
40 view.restore_consume(consume);
41 ctx->signal<Ev>(sequence);
42 }
43 else if constexpr (Ev == Event::KeyMouseInput)
44 {
45 ctx->signal<Ev>(ref(*input()));
46 }
47 else if constexpr (Ev == Event::MouseMoved)
48 {
49 ctx->signal<Ev>(input()->mouse_position());
50 }
51 else if constexpr (Ev == Event::MouseInput)

53 ctx->signal<Ev>(ref(*input()));
54 }
55 else if constexpr (Ev == Event::Resize)
56 {
57 auto root = _ts.current->state->root();
58 ctx->signal<Ev>(root->box());
59 }
60 else
61 ctx->signal<Ev>(std::forward<Argv>(args)...);

331 std::chrono::milliseconds SystemScreen::_run_animations()
332 {
333 std::chrono::milliseconds deadline{std::chrono::milliseconds::max()};
334 auto& interps = _ts.current->animations;
335 if (!interps.empty())
336 {
337 const auto end = interps.end();
338 const auto beg = interps.begin();
339 for (auto it = beg; it != end;)
340 {
341 auto& [anim, callback] = it->second;
342 if (!anim->keep_alive())
343 {
344 const auto saved = it;
345 ++it;
346 auto cb = std::move(callback);
347 auto target = d2::TreeIter{anim->target().first.lock()};
348 interps.erase(saved);
349 D2_SAFE_BLOCK_BEGIN
350 cb(target);
351 D2_SAFE_BLOCK_END
352 if (it == interps.end())
353 break;
354 }
355 else
356 {
357 deadline = std::min(deadline, anim->update());
358 ++it;
359 }
360 }
361 }
362 return deadline;

614 void SystemScreen::set(std::string_view name)
615 {
616 D2_TLOG(Info, "Switching trees to: '", name, "'")
617 if (_ts.current != nullptr && name == root()->name())
618 return;
619
620 const auto prev = _ts.current_name;
621 auto root = _trees[name];
622 if (_ts.current != nullptr)
623 {
624 if (_ts.current->tags.tag<bool>("SwapOut"))
625 {
626 D2_TLOG(Verbose, "Swapping current tree out")
627 _ts.current->state->swap_out();
628 this->root()->setstate(Element::Swapped, true);
629 _ts.current->swapped_out = true;
630 }
631 if (_ts.current->tags.tag<bool>("SwapClean"))
632 {
633 D2_TLOG(Verbose, "Clearing current tree out")
634 _ts.current->state->root()->clear();
635 _ts.current->unbuilt = true;
636 }
637 if (const auto value =
638 _ts.current->tags.tag<std::chrono::milliseconds>("OverrideRefresh");
639 value != std::chrono::milliseconds(0))
640 {
641 D2_TLOG(Verbose, "Restoring refresh rate to: ", _restore_refresh)
642 set_refresh_rate(_restore_refresh);
643 }
644 }
645 _ts.current_name = name;
646 _ts.current = root;
647 _ts.focused = nullptr;
648
649 if (_ts.current->unbuilt)
650 {
651 D2_TLOG(Verbose, "Building new tree")
652 _ts.current->rebuild(this->root(), _ts.current->state);
653 _ts.current->unbuilt = false;
654 }
655 if (_ts.current->swapped_out)
656 {
657 D2_TLOG(Verbose, "Swapping in tree")
658 root->state->swap_in();
659 this->root()->setstate(Element::Swapped, false);
660 _ts.current->swapped_out = false;
661 }
662
663 if (const auto value = _ts.current->tags.tag<std::chrono::milliseconds>("OverrideRefresh");
664 value != std::chrono::milliseconds(0))
665 {
666 D2_TLOG(Verbose, "Overriding refresh rate to: ", value)
667 _restore_refresh = _refresh_rate;
668 set_refresh_rate(value);
669 }
670
671 D2_TLOG(Verbose, "Recycling")
672 context()->input().ptr()->begincycle();
673 context()->input().ptr()->endcycle();
674 _update_viewport();
675
676 D2_TLOG(Verbose, "Initializing tree")
677 root->state->root()->initialize(true);
678
679 _signal<Event::TreeSwap>(std::string(prev), std::string(name));
680 }

834 void SystemScreen::tick()
835 {
836 const auto beg = std::chrono::steady_clock::now();
837 const auto ctx = context();
838 const auto input = ctx->input();
839 std::chrono::steady_clock::time_point interp_deadline{
840 std::chrono::steady_clock::time_point::max()
841 };
842 // Update
843 {
844 root()->setcache(d2::Element::CachePolicy::Static);
845 if (input->had_event(in::Event::ScreenResize))
846 {
847 _update_viewport();
848 }
849 if (input->had_event(in::Event::MouseMovement) ||
850 input->had_event(in::Event::KeyMouseInput))
851 {
852 const auto target = _update_states(root(), ctx->input().ptr()->mouse_position());
853 const auto is_released = input->active(in::mouse::Left, in::mode::Release);
854 const auto is_pressed = input->active(in::mouse::Left, in::mode::Press);
855 const auto cpy_targetted = _ts.targetted;
856
857 auto uptarget = target;
858 if (target != nullptr && is_pressed && !target->provides_cursor_sink() &&
859 target->parent() != nullptr)
860 {
861 uptarget = _update_states_reverse(target->parent());
862 }
863
864 auto _1 = _ts.targetted.shared();
865 auto _2 = _ts.clicked.shared();
866 auto _3 = _ts.focused.shared();
867 if (_ts.targetted != target)
868 {
869 if (_ts.targetted != nullptr)
870 _ts.targetted->setstate(Element::State::Hovered, false);
871
872 if (target != nullptr)
873 target->setstate(Element::State::Hovered, true);
874
875 _ts.targetted = target;
876 _trigger_rc_hover_events(target, cpy_targetted);
877 }
878 if (is_pressed)
879 {
880 const auto old_clicked = _ts.clicked;
881 if (_ts.clicked != uptarget)
882 {
883 if (_ts.clicked != nullptr)
884 _ts.clicked->setstate(Element::State::Clicked, false);
885 _ts.clicked = uptarget;
886 if (_ts.clicked != nullptr)
887 _ts.clicked->setstate(Element::State::Clicked, true);
888 _trigger_rc_click_events(_ts.clicked, old_clicked);
889 }
890 else if (_ts.clicked != nullptr)
891 {
892 _ts.clicked->setstate(Element::State::Clicked, true);
893 }
894 }
895 if (is_released)
896 {
897 const auto old_clicked = _ts.clicked;
898
899 if (_ts.clicked != nullptr)
900 _ts.clicked->setstate(Element::State::Clicked, false);
901 _ts.clicked = nullptr;
902 _trigger_rc_click_events(nullptr, old_clicked);
903 }
904 if (_ts.focused != uptarget && is_pressed)
905 {
906 if (_ts.keynav_iterator != nullptr)
907 _ts.keynav_iterator->setstate(Element::State::Keynavi, false);
908 focus(uptarget);
909 }
910 }
911
912 _ts.current->state->update();
913
914 _trigger_events();
915
916 const auto interp_ms = _run_animations();
917 ctx->deadline(interp_ms == Animation::Wake::refresh ? _refresh_rate : interp_ms);
918
919 if (input->had_pulse())
920 ctx->deadline(std::chrono::milliseconds{0});
921 }
922 // Render
923 if (!ctx->is_suspended() && root()->needs_update())
924 {
925 _signal<Event::PreRedraw>();
926
927 auto& root = *this->root().as();
928 auto frame = root.frame();
929
930 const auto [bwidth, bheight] = root.box();
931 auto output = ctx->output().ptr();
932 output->write_frame(frame.buffer());
933
934 _signal<Event::PostRedraw>();
935 }
936 {
937 _fps_ctr++;
938 if (beg - _last_mes > std::chrono::seconds(1))
939 {
940 _fps_avg = _fps_ctr;
941 _fps_ctr = 0;
942 _last_mes = std::chrono::steady_clock::now();
943 }
944
945 const auto end = std::chrono::steady_clock::now();
946 const auto delta = end - beg;
947 _prev_delta = std::chrono::duration_cast<std::chrono::microseconds>(delta);
948
949 if (_refresh_rate != std::chrono::milliseconds::max())
950 ctx->deadline(
951 _refresh_rate - std::chrono::duration_cast<std::chrono::milliseconds>(delta)
952 );
953 }
954 }
