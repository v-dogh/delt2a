738 constexpr void blend(combined src) noexcept
739 {
740 constexpr auto inv = 1.0f / 255.0f;
741
742 const auto src_fa = src.af;
743
744 if (src_fa == 0 && src.a == 0)
745 return;
746
747 if (src_fa == 255)
748 {
749 rf = src.rf;
750 gf = src.gf;
751 bf = src.bf;
752 af = 255;
753 }
754 else if (v == ' ')
755 {
756 const auto alpha_src = src_fa * inv;
757 rf = impl::blend(src.rf, r, alpha_src);
758 gf = impl::blend(src.gf, g, alpha_src);
759 bf = impl::blend(src.bf, b, alpha_src);
760 af = 255;
761 }
762 else if (src.af != 0)
763 {
764 const auto alpha_src = src_fa * inv;
765 const auto alpha_dest = af * inv;
766
767 rf = impl::blend(src.rf, rf, alpha_src);
768 gf = impl::blend(src.gf, gf, alpha_src);
769 bf = impl::blend(src.bf, bf, alpha_src);
770 af = static_cast<component>(
771 (alpha_src + alpha_dest * (1.0f - alpha_src)) * 255.0f + 0.5f
772 );
773 }
774
775 if (src.a == 255)
776 {
777 r = src.r;
778 g = src.g;
779 b = src.b;
780 a = 255;
781 }
782 else if (src.a != 0)
783 {
784 const auto alpha_src = src.a * inv;
785 const auto alpha_dest = a * inv;
786
787 r = impl::blend(src.r, r, alpha_src);
788 g = impl::blend(src.g, g, alpha_src);
789 b = impl::blend(src.b, b, alpha_src);
790 a = static_cast<component>(
791 (alpha_src + alpha_dest * (1.0f - alpha_src)) * 255.0f + 0.5f
792 );
793 }
794
795 if ((v == ' ' && src_fa > 0) || src_fa >= af)
796 {
797 v = src.v;
798 style = src.style;
799 }
800 }
