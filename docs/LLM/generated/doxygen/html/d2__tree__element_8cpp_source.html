564 Element::Frame Element::frame()
565 {
566 if (_internal_state & WasWrittenLayout)
567 {
568 _update_layout_impl();
569 _signal_update(WasWrittenLayout);
570 }
571 if (_internal_state & CachePolicyBypass)
572 {
573 if (needs_update() || _buffer.empty())
574 {
575 if (const auto [width, height] = box(); width > 0 && height > 0)
576 {
577 _update_style_impl();
578 const auto [bwidth, bheight] = _reserve_buffer_impl();
579 _buffer.set_size(bwidth, bheight);
580 _internal_state |= IsBeingRendered;
581 _signal_update(WasWritten);
582 _frame_impl(_fetch_pixel_buffer_impl());
583 _internal_state &= ~IsBeingRendered;
584 }
585 else
586 {
587 _signal_update(WasWritten);
588 _buffer.clear();
589 }
590 }
591 }
592 else
593 {
594 if (_internal_state & WasWrittenLayout)
595 {
596 _update_layout_impl();
597 _signal_update(WasWrittenLayout);
598 }
599 if (needs_update() || _buffer.empty())
600 {
601 if (const auto [width, height] = box(); width > 0 && height > 0)
602 {
603 const auto ab = !_provides_buffer_impl() || parent() == nullptr;
604 _frames_reused = 0;
605 _update_style_impl();
606 bool isolated = parent() == nullptr;
607 if (auto owner = parent())
608 {
609 const auto target = owner->_fetch_pixel_buffer_impl();
610 const auto pos = position();
611 isolated = target.has_images() || pos.x < 0 || pos.y < 0 ||
612 static_cast<long long>(pos.x) + width > target.width() ||
613 static_cast<long long>(pos.y) + height > target.height();
614 }
615 if (isolated || (((_internal_state & CachePolicyStatic) ||
616 ((_internal_state & CachePolicyDynamic) && _frames_redrawn == 1)) && ab))
617 {
618 const auto [bwidth, bheight] = _reserve_buffer_impl();
619 _buffer.set_size(bwidth, bheight);
620 _frames_redrawn = 0;
621 }
622 else
623 {
624 _buffer.clear();
625 _frames_redrawn++;
626 }
627
628 D2_ASSERT(!(_buffer.empty() && parent() == nullptr))
629 _internal_state |= IsBeingRendered;
630 _signal_update(WasWritten);
631 _frame_impl(_fetch_pixel_buffer_impl());
632 _internal_state &= ~IsBeingRendered;
633
634 if ((_internal_state & CachePolicyStatic) && !_buffer.empty())
635 {
636 _buffer.compress();
637 _frames_reused = 10;
638 }
639 }
640 else
641 {
642 _signal_update(WasWritten);
643 _buffer.clear();
644 }
645 }
646 else if (!(_buffer.empty() || _buffer.compressed()) && ++_frames_reused == 10)
647 {
648 _buffer.compress();
649 }
650 else
651 _frames_reused = 0;
652 }
653
654 return Frame(shared_from_this());
655 }
