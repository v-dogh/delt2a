437 SignalStorage& storage,
438 event_idx ev,
439 const SpecifierKey& dispatch_specifier,
440 SignalInstance& args
441 )
442 {
443 std::vector<std::shared_ptr<HandleState>> callbacks;
444
445 auto collect = [&](SignalStorage::HandleList& list)
446 {
447 list.erase(
448 std::remove_if(
449 list.begin(),
450 list.end(),
451 [](const std::shared_ptr<HandleState>& ptr)
452 {
453 return ptr == nullptr || ptr->state.load(std::memory_order::acquire) ==
454 Handle::State::Inactive;
455 }
456 ),
457 list.end()
458 );
459
460 callbacks.reserve(callbacks.size() + list.size());
461
462 for (const auto& ptr : list)
463 {
464 if (ptr == nullptr)
465 continue;
466
467 const auto state = ptr->state.load(std::memory_order::acquire);
468 if (state == Handle::State::Active)
469 callbacks.push_back(ptr);
470 }
471 };
472
473 {
474 std::lock_guard lock(storage.mtx);
475
476 auto f = storage.handles.find(ev);
477 if (f == storage.handles.end())
478 return;
479
480 collect(f->second.generic);
481
482 if (!dispatch_specifier.empty())
483 {
484 auto keyed = f->second.keyed.find(dispatch_specifier);
485 if (keyed != f->second.keyed.end())
486 {
487 collect(keyed->second);
488 if (keyed->second.empty())
489 f->second.keyed.erase(keyed);
490 }
491 }
492
493 if (f->second.generic.empty() && f->second.keyed.empty())
494 storage.handles.erase(f);
495 }
496
497 for (const auto& ptr : callbacks)
498 {
499 if (ptr->state.load(std::memory_order::acquire) == Handle::State::Active)
