173 template<typename Func, typename... Argv>
174 auto sync_async(Func&& callback, Argv&&... args)
175 -> mt::future<std::invoke_result_t<Func&, Argv...>>
176 {
177 using ret = std::invoke_result_t<Func&, Argv...>;
178 auto task = mt::future<ret>::make();
179 auto run = [callback = std::forward<Func>(callback),
180 ... args = std::forward<Argv>(args), task]() mutable
181 {
182 if (task.is_discarded())
183 return;
184 try
185 {
186 if constexpr (std::is_void_v<ret>)
187 {
188 callback(std::forward<Argv>(args)...);
189 task.block()->set_value();
190 }
191 else
192 task.block()->set_value(callback(std::forward<Argv>(args)...));
193 }
194 catch (...)
195 {
196 task.block()->set_exception(std::current_exception());
197 }
198 };
199 if (is_synced())
200 run();
201 else
202 {
203 if (!_worker)
204 throw std::logic_error("IO context has no main worker");
205 _worker->accept({[run = std::move(run)](mt::Task::Query query, std::any& out) mutable
206 {
207 if (query == mt::Task::Query::Type)
208 out = static_cast<unsigned char>(mt::Task::Immediate);
209 else if (query == mt::Task::Query::Run)
210 {
211 run();
212 out = mt::Task::Token::Discard;
213 }
214 }});
215 }
216 return task;
217 }
