129 int ParentElement::resolve_units(Unit value, cptr elem) const
130 {
131 const bool adjust = value.getmods() & Unit::Mods::Adjusted;
132 const bool dims = value.getmods() & std::uint8_t(UnitContext::Dimensions);
133 const bool horiz = value.getmods() & std::uint8_t(UnitContext::Horizontal);
134 const bool inv = value.getmods() & Unit::Mods::Inverted;
135 const bool rel = value.getmods() & Unit::Relative;
136 const float val = value.raw();
137
138 const auto border_top = _get_border_impl(BorderType::Top, elem) * (!rel || dims);
139 const auto border_bottom = _get_border_impl(BorderType::Bottom, elem) * (!rel || dims);
140 const auto border_left = _get_border_impl(BorderType::Left, elem) * (!rel || dims);
141 const auto border_right = _get_border_impl(BorderType::Right, elem) * (!rel || dims);
142 const auto border_horizontal = border_left + border_right;
143 const auto border_vertical = border_top + border_bottom;
144 const auto border_off = dims ? 0
145 : (horiz ? (inv ? -border_right : border_left)
146 : (inv ? -border_bottom : border_top));
147
148 int result = 0;
149 if ((value.getmods() & Unit::Center) && !dims)
150 {
151 const auto ww = elem->layout(Layout::Width);
152 const auto pw = layout(Layout::Width);
153 result =
154 ((horiz
155 ? (layout(Layout::Width) - elem->layout(Layout::Width) - border_horizontal)
156 : (layout(Layout::Height) - elem->layout(Layout::Height) - border_vertical)) /
157 2) +
158 border_off +
159 resolve_units(
160 Unit(
161 value.raw(),
162 value.getunits(),
163 value.getmods() & ~Unit::Center | Unit::Relative
164 ),
165 elem
166 );
167 }
168 else
169 {
170 switch (value.getunits())
171 {
172 case Unit::Auto:
173 {
174 Unit cpy = value;
175 if (dims)
176 {
177 cpy.setunits(Unit::Pc);
178 cpy = 1.f - val;
179 }
180 else
181 {
182 cpy.setunits(Unit::Px);
183 }
184 result = resolve_units(cpy, elem);
185 break;
186 }
187 case Unit::Px:
188 {
189 if (inv)
190 {
191 if (dims)
192 {
193 result = (horiz ? (layout(Layout::Width) - border_horizontal)
194 : (layout(Layout::Height) - border_vertical)) -
195 val;
196 }
197 else
198 {
199 result =
200 ((horiz ? (layout(Layout::Width) - elem->layout(Layout::Width))
201 : (layout(Layout::Height) - elem->layout(Layout::Height))) -
202 val) +
203 border_off;
204 }
205 }
206 else
207 result = val + border_off;
208 break;
209 }
210 case Unit::Pv:
211 {
212 const auto input = context()->input();
213 const auto [width, height] = input->screen_size();
214 const auto mval = (inv ? (1 - val) : val) - dims;
215 result = std::ceil((mval * (horiz ? width : height)) + border_off);
216 break;
217 }
218 case Unit::Pc:
219 {
220 const auto mval = inv ? (1 - val) : val;
221 result = std::ceil(
222 (mval * (horiz ? (layout(Layout::Width) - border_horizontal)
223 : (layout(Layout::Height) - border_vertical))) +
224 border_off
225 );
226 break;
227 }
228 }
229 }

231 if (adjust)

236 }
