242 TopLeft = 1 << 0,
243 TopRight = 1 << 3,
244 MidTopLeft = 1 << 1,

247 MidBotRight = 1 << 5,

251 static std::string cell(uint8_t mask)
252 {
253 const char32_t ch = 0x2800 + mask;
254 std::string s;
255 s.reserve(3);
256 if (ch <= 0xFFFF)
257 {
258 s.push_back(char(0xE0 | ((ch >> 12) & 0x0F)));
259 s.push_back(char(0x80 | ((ch >> 6) & 0x3F)));
260 s.push_back(char(0x80 | (ch & 0x3F)));
261 }
262 return s;
263 }
264 };
