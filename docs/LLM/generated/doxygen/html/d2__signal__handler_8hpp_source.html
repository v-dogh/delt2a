396 // It supports basically two classes of identifiers + three identifier types
397 // We have signal identifiers and bucket identifiers
398 // For the signal IDs we got:
399 // 1. Event enum type
400 // 2. <optional> Signal specifier type (runtime)
401 // And for the buckets we have (these simply sort inside a specific signal):

629 auto aligned_size = _static_storage.size();
630 if (const auto ptr = std::align(alignment, total, aligned, aligned_size); ptr)
631 {
632 new (ptr) type{std::forward<Argv>(pack)...};
633 _manager = +[](SignalInstance* ptr,
634 SignalInstance* dest,
635 void* callback,
636 void* block,
637 Action action) -> void*
638 {
639 auto aligned = static_cast<void*>(ptr->_static_storage.data());
640 auto aligned_size = ptr->_static_storage.size();
641 const auto data = std::align(alignment, total, aligned, aligned_size);
642
643 if (action == Action::Destroy)
644 {
645 std::destroy_at(reinterpret_cast<type*>(data));
646 }
647 else if (action == Action::Ptr)
648 {

650 }

658 dest->_manager = ptr->_manager;
659 ptr->_manager = nullptr;
660 }
661 else if (action == Action::Apply)
662 {
663 auto* args = reinterpret_cast<type*>(data);
664 auto* cb = *reinterpret_cast<func*>(callback);
665 std::apply([&](auto&... args) { cb(block, args...); }, *args);
666 }

668 return nullptr;
669 };
670 }
