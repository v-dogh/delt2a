297 {

304 new_view.resize(view_count);
305

309 if (idx < 0 || idx >= int(data::length))

311 new_view[i] = nullptr;

313 }

343 }

348 void DemandScrollBox::_frame_impl(PixelBuffer::View buffer)
349 {
350 _update_view();
351
352 buffer.fill(pixel::combine(data::foreground_color, data::background_color));
353
354 // Render elements
355 const auto bw =
356 (data::container_options & EnableBorder) ? resolve_units(data::border_width) : 0;
357 const auto height = layout(d2::Element::Layout::Height) - bw * 2;
358 const auto xpad = resolve_units(data::padding_horizontal);
359 const auto ypad = resolve_units(data::padding_vertical);
360 std::size_t logical_height = 0;
361 if (!_view.empty())
362 {
363 for (std::size_t i = data::preload; i < _view.size() - data::preload; i++)
364 {
365 logical_height += ypad;

367 const auto frame = it->frame();
368 const auto [x, y] = it->position();
369 const auto [width, height] = it->box();
370 it->override_position(x + xpad, logical_height);
371 buffer.inscribe(x + xpad, logical_height, frame.buffer());
372 logical_height += height;
373 }
374 }
375 // Render scrollbar
376 {
377 // Update the slider width
378 const auto logical_count = _view.size() - (data::preload * 2);
379 const auto slider_height = layout(Element::Layout::Height);
380 const auto sw = _scrollbar->get<Slider::SliderWidth>();
381 const auto sm = _scrollbar->get<Slider::Max>();
382 const auto est_height = ((float(logical_height) / logical_count) * data::length);
383 const auto rsw =
384 std::max(1, std::min<int>(slider_height, slider_height * height / est_height));
385 // Update slider and render scrollbar
386 if (est_height > height || !data::hide_slider)
387 {
388 _scrollbar->setstate(Display, true);
389 if (sw != rsw)
390 _scrollbar->set<Slider::SliderWidth>(rsw);
391 if (sm != est_height - height)
392 _scrollbar->set<Slider::Max>(est_height - height);
393 const auto frame = _scrollbar->frame();
394 const auto [x, y] = _scrollbar->position();
395 buffer.inscribe(x, y, frame.buffer());
396 }

398 _scrollbar->setstate(Display, false);
