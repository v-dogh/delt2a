94 enum InternalState : internal_flag
95 {
96 IsBeingRendered = 1 << 0,
97 // Set if the element is being initialized (prevents write signal propagation etc.)
98 IsBeingInitialized = 1 << 1,
99 // Set if the style is changed (i.e. requires a redraw)
100 WasWritten = 1 << 2,
101 // Set if the layout of the object has changed (for parent elements)
102 WasWrittenLayout = 1 << 3,
103 // Set when the x-position of the object was updated
104 PositionXUpdated = 1 << 4,
105 // Set when the y-position of the object was updated
106 PositionYUpdated = 1 << 5,
107 // Set when the width of the bounding box of the object was updated
108 DimensionsWidthUpdated = 1 << 6,
109 // Set when the height of the bounding box of the object was updated
110 DimensionsHeightUpdated = 1 << 7,
111 // Indicates that the object's framebuffer should not be compressed
112 CachePolicyDynamic = 1 << 8,
113 // Indicates that the object should me drawn in immediate mode
114 CachePolicyVolatile = 1 << 9,
115 // Indicates that the object's framebuffer should be preemptively compressed
116 CachePolicyStatic = 1 << 10,
117 // Bypasses any caching techniques
118 CachePolicyBypass = 1 << 11,
119 };
