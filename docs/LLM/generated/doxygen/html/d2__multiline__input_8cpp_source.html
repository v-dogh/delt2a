123 MultiInput::Viewport MultiInput::_viewport(const DocModel& doc) const
124 {
125 const auto [boxw, boxh] = box();
126
127 Viewport vp{std::size_t(std::max(0, boxw)), std::size_t(std::max(0, boxh)), false, false};
128
129 const auto allow_v = data::enable_scrollbar_vertical && _fixed_height();
130 const auto allow_h = data::enable_scrollbar_horizontal && _fixed_width();
131
132 if (!allow_v && !allow_h)
133 return vp;
134
135 const auto vbar_w = std::size_t(_vscroll_width());
136 const auto hbar_h = std::size_t(_hscroll_height());
137 const auto render_width = _render_max_width(doc);
138
139 for (int i = 0; i < 4; i++)
140 {
141 const auto avail_w = std::max<std::size_t>(1, vp.width);
142 const auto avail_h = std::max<std::size_t>(1, vp.height);
143
144 const auto next_v = allow_v && (_line_count(doc) > avail_h);
145 const auto next_h = allow_h && (render_width > avail_w);
146
147 if (next_v == vp.show_vscroll && next_h == vp.show_hscroll)
148 break;
149
150 vp.show_vscroll = next_v;
151 vp.show_hscroll = next_h;
152
153 vp.width = std::size_t(std::max(0, boxw - int(vp.show_vscroll ? vbar_w : 0)));
154 vp.height = std::size_t(std::max(0, boxh - int(vp.show_hscroll ? hbar_h : 0)));
155 }

880 void MultiInput::_frame_impl(PixelBuffer::View buffer)
881 {
882 buffer.fill(data::background_color);
883 if (buffer.empty())
884 return;
885 _clamp_ptr();
886 const auto color = pixel::combine(data::foreground_color, data::background_color);
887 const auto& doc = _doc_model();
888 const auto vp = _viewport(doc);
889 auto content = PixelBuffer::View(buffer, 0, 0, int(vp.width), int(vp.height));
890 const auto& layout = doc.text.display();
891 const auto gap = _draw_ptr() && !_ptr_overlapped();
892 layout.draw(content, color, {-int(_view_x), -int(_view_y)},
893 text::Range{_highlight_start, _highlight_end}, data::highlight_mask,
894 gap ? std::optional(_ptr_abs) : std::nullopt);
895 if (!doc.length && !_draw_ptr())
896 text::Layout(data::hint).draw(content,
897 pixel::combine(data::foreground_color.alpha(0.5f).stylize(px::style::Bold), data::background_color));
898 const auto position = layout.caret(_ptr_abs) - Position{int(_view_x), int(_view_y)};
899 if (_draw_ptr() && position.x >= 0 && position.x < int(vp.width) &&
900 position.y >= 0 && position.y < int(vp.height))
901 {
902 if (!gap && _ptr_abs < layout.size())
903 for (int i = 0; i < layout.glyphs()[_ptr_abs].columns && position.x + i < int(vp.width); ++i)
904 content.at(position.x + i, position.y) = pixel::combine(data::foreground_color.extend(' '), data::background_color);
905 _write_ptr(content.at(position.x, position.y));
906 }
907
908 if (vp.show_vscroll && _vscrollbar != nullptr)
909 {
910 const auto frame = _vscrollbar->frame();
911 const auto [x, y] = _vscrollbar->position();
912 buffer.inscribe(x, y, frame.buffer());
913 }
914
915 if (vp.show_hscroll && _hscrollbar != nullptr)
916 {
917 const auto frame = _hscrollbar->frame();
918 const auto [x, y] = _hscrollbar->position();
919 buffer.inscribe(x, y, frame.buffer());
920 }
