117 void SystemModule::load()
118 {
119 _transition(true,
120 [this]()
121 {
122 D2_TAG_MODULE_RUNTIME(info().name)
123 auto stat = _pre_load_impl();
124 if (stat != Status::Ok)
125 return stat;
126 stat = _load_impl();
127 if (stat == Status::Offline)

129 D2_TLOG(Warning, "Module degraded")

131 }
132 else
133 D2_TLOG(Module, "Loaded module")

136 );
