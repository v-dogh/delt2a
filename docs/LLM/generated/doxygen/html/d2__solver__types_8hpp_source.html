195 {
196 private:
197 std::size_t _size_vars{ 0 };
198 std::size_t _size_locals{ 0 };
199 std::size_t _size_globals{ 0 };
200 std::array<Variable, 24> _terms{};
201 Constraint _op{ Constraint::Equal };
202 result_type _constant{ 0 };
203 result_type _scale{ 0 };
204
205 static constexpr void _accumulate_expr(
206 std::span<Variable> sink_v,
207 std::span<Variable> sink_l,
208 std::span<Variable> sink_g,
209 const Expression& source,
210 Expression& dest,
211 int sign
212 )
213 {
214 std::size_t vcnt = 0;
215 std::size_t lcnt = 0;
216 std::size_t gcnt = 0;
217
218 std::span<Variable> vs = sink_v.subspan(dest._size_vars);
219 std::span<Variable> ls = sink_l.subspan(dest._size_locals);
220 std::span<Variable> gs = sink_g.subspan(dest._size_globals);
221
222 for (std::size_t i = 0; i < source._size_vars; i++)
223 {
224 const auto& v = source._terms[i];
225 const auto type = v.type();
226 const auto c = v.coefficient() * sign;
227 const auto var = Variable(v, c);
228 dest._scale += std::abs(c);
229 switch (type)
230 {
231 case VariableIndex::Type::Constant: dest._constant += c; break;
232 case VariableIndex::Type::Variable: vs[vcnt++] = var; dest._size_vars++; break;
233 case VariableIndex::Type::Local: ls[lcnt++] = var; dest._size_locals++; break;
234 case VariableIndex::Type::Global: gs[gcnt++] = var; dest._size_globals++; break;
235 default: std::abort();
236 }
237 }
238 }
239 public:
240 // Returns an expression in form
241 // _terms[0] + ... + _terms[n] + _constant <_op> 0
242 // Performs normalization if the operator is <more/eq>
243 static constexpr Expression reduce(const Expression& lhs, const Expression& rhs, Constraint op)
244 {
245 Expression result;
246 result._op = op;
247
248 std::array<Variable, 24> vars;
249 std::array<Variable, 24> locals;
250 std::array<Variable, 24> globals;
251
252 // Accumulate types into their respective storage
253 // (coefficients are added directly to result._rhs)
254
255 _accumulate_expr(vars, locals, globals, lhs, result, 1);
256 _accumulate_expr(vars, locals, globals, rhs, result, -1);
257
258 // Simplify and write back
259
260 std::copy(
261 vars.begin(),
262 vars.begin() + result._size_vars,
263 result._terms.begin()
264 );
265 std::copy(
266 locals.begin(),
267 locals.begin() + result._size_locals,
268 result._terms.begin() + result._size_vars
269 );
270 std::copy(
271 globals.begin(),
272 globals.begin() + result._size_globals,
273 result._terms.begin() + result._size_vars + result._size_locals
274 );
275
276 // Normalize
277
278 // We add an epsilon (and transform into a less/more/eq inequality)
279 if (op == Constraint::More || op == Constraint::Less)
280 {
281 constexpr auto epsilon_rel = result_type(1e-8);
282 constexpr auto epsilon_abs = result_type(1e-7);
283 const auto epsilon = std::max(epsilon_abs, epsilon_rel * std::max(result_type(1), result._scale));
284 result._constant -= epsilon;
285 if (op == Constraint::More) op = Constraint::MoreEq;
286 else if (op == Constraint::Less) op = Constraint::LessEq;
287 }
288
289 // Transform into a less/eq inequality
290 const auto is_more = op == Constraint::MoreEq;
291 if (is_more)
292 {
293 for (std::size_t i = 0; i < result._size_globals + result._size_locals + result._size_vars; i++)
294 {
295 auto& term = result._terms[i];
296 term = Variable(term, term.coefficient() * -1.f);
297 }
298 result._constant *= -1.f;
299 result._op = Constraint::LessEq;
300 }
301
302 return result;
303 }
