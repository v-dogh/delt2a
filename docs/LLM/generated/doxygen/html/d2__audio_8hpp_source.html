234 {
235 auto phase = 0.f;
236 const auto phase_inc = frequency / sample_rate;
237 return [phase, phase_inc](float) mutable
238 {
239 const auto out = std::sin(phase * 2.f * 3.1415926535f);
240 phase += phase_inc;
241 if (phase >= 1.f)
242 phase -= 1.f;

246 static auto square(float frequency = 440.f, float sample_rate = 48'000.f)
247 {
248 auto phase = 0.f;
249 const auto phase_inc = frequency / sample_rate;
250 return [phase, phase_inc](float) mutable
251 {
252 const auto out = (phase < 0.5f) ? 1.f : -1.f;
253 phase += phase_inc;
254 if (phase >= 1.f)
255 phase -= 1.f;

257 };

259 static auto saw(float frequency = 440.f, float sample_rate = 48'000.f)
260 {
261 auto phase = 0.f;
262 const auto phase_inc = frequency / sample_rate;
263 return [phase, phase_inc](float) mutable
264 {
265 const auto out = 2.f * phase - 1.f;
266 phase += phase_inc;
267 if (phase >= 1.f)
268 phase -= 1.f;
269 return out;
270 };
271 }
272 static auto triangle(float frequency = 440.f, float sample_rate = 48'000.f)
273 {
274 auto phase = 0.f;
275 const auto phase_inc = frequency / sample_rate;
276 return [phase, phase_inc](float) mutable
277 {
278 float out = 0.f;
279 if (phase < 0.5f)
280 out = 4.f * phase - 1.f;
281 else
282 out = 3.f - 4.f * phase;
283
284 phase += phase_inc;
285 if (phase >= 1.f)
286 phase -= 1.f;
287 return out;
288 };
