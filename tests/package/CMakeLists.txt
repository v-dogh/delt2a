cmake_minimum_required(VERSION 3.21)
project(DeltaAsciiPackageConsumer LANGUAGES CXX)
find_package(DeltaAscii CONFIG REQUIRED)
add_executable(package_consumer ../consumer.cpp)
target_link_libraries(package_consumer PRIVATE DeltaAscii::Core)
target_compile_definitions(package_consumer PRIVATE D2_PACKAGE_EDITORS)

if (TARGET DeltaAscii::ImagesKitty)
    target_compile_definitions(package_consumer PRIVATE D2_PACKAGE_IMAGES)
endif()
if (TARGET DeltaAscii::Inference)
    target_link_libraries(package_consumer PRIVATE DeltaAscii::Inference)
    target_compile_definitions(package_consumer PRIVATE D2_PACKAGE_INFERENCE)
endif()
if (TARGET DeltaAscii::OpenAI)
    target_link_libraries(package_consumer PRIVATE DeltaAscii::OpenAI)
    target_compile_definitions(package_consumer PRIVATE D2_PACKAGE_OPENAI)
endif()

if (TARGET DeltaAscii::PngLoader)
    target_link_libraries(package_consumer PRIVATE DeltaAscii::PngLoader)
    target_compile_definitions(package_consumer PRIVATE D2_PACKAGE_PNG)
endif()
